\documentclass[tikz]{standalone}

\usepackage{pgfplotstable}
\pgfplotsset{compat=1.18}

\begin{document}

\pgfplotstableread{\jobname.dat}\newdata
\pgfplotstablegetcolsof{\newdata}
\pgfmathtruncatemacro\NoOfCols{\pgfplotsretval-1}

\begin{tikzpicture}[scale = 2]
  \begin{semilogyaxis}[
    height = 5cm, width = 7.5cm, scale only axis, xlabel = { $k\;(m = 128)$ },
    ylabel = { CPU cycles }, cycle list name = color list,
    legend pos = outer north east,
    legend style = {
      legend columns = 1, font = \small, draw = none, fill = none,
      legend cell align = left, legend image code/.code = {
        \draw[mark repeat = 2, mark phase = 2]
          plot coordinates { (0cm, 0cm) (0.1cm, 0cm) (0.2cm, 0cm) };
      },
    },
  ]

  \pgfplotsinvokeforeach{1,...,\NoOfCols}{
    \pgfplotstablegetcolumnnamebyindex{#1}\of{\newdata}\to{\colname}
    \addplot+[mark = none, thick] table [y index = #1] {\newdata};
    \addlegendentryexpanded{\colname}
  }
  \end{semilogyaxis}
\end{tikzpicture}

\end{document}
